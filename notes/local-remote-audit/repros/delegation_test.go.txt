package main

import (
	"bufio"
	"encoding/json"
	"github.com/flopwire/flopwire/internal/agent"
	"github.com/flopwire/flopwire/internal/localindex"
	"io"
	"log/slog"
	"net"
	"os"
	"path/filepath"
	"testing"
)

func TestAuditOnceIgnoresRequestedRetention(t *testing.T) {
	for _, heldSyncOnly := range []bool{false, true} {
		t.Run(map[bool]string{false: "full_to_sync_only", true: "sync_only_to_full"}[heldSyncOnly], func(t *testing.T) {
			dir := t.TempDir()
			db := filepath.Join(dir, "index.db")
			socketDir, err := os.MkdirTemp("/tmp", "fw-as-")
			if err != nil {
				t.Fatal(err)
			}
			defer os.RemoveAll(socketDir)
			sock := filepath.Join(socketDir, "a.sock")
			held, err := localindex.Open(db, localindex.Options{SyncOnly: heldSyncOnly})
			if err != nil {
				t.Fatal(err)
			}
			defer held.Close()
			ln, err := net.Listen("unix", sock)
			if err != nil {
				t.Fatal(err)
			}
			defer ln.Close()
			got := make(chan agent.Request, 1)
			go func() {
				c, e := ln.Accept()
				if e != nil {
					return
				}
				defer c.Close()
				line, _ := bufio.NewReader(c).ReadBytes('\n')
				var req agent.Request
				json.Unmarshal(line, &req)
				got <- req
				c.Write([]byte("{\"ok\":true}\n"))
			}()
			s, err := openAgentIndex(t.Context(), db, localindex.Options{SyncOnly: !heldSyncOnly}, true, sock, slog.New(slog.NewTextHandler(io.Discard, nil)))
			if err != nil || s != nil {
				t.Fatalf("unexpected result %v %v", s, err)
			}
			req := <-got
			if req.Op != "pass" {
				t.Fatalf("request %+v", req)
			}
			var mode string
			if err := held.DB().QueryRow("SELECT value FROM meta WHERE key='mode'").Scan(&mode); err != nil {
				t.Fatal(err)
			}
			want := map[bool]string{false: localindex.ModeFull, true: localindex.ModeSyncOnly}[heldSyncOnly]
			if mode != want {
				t.Fatalf("mode %q want %q", mode, want)
			}
			t.Logf("Requested SyncOnly=%v returned success, existing mode=%s; request=%+v", !heldSyncOnly, mode, req)
		})
	}
}
