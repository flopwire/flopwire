package main

import (
	"github.com/flopwire/flopwire/internal/localindex"
	"github.com/flopwire/flopwire/internal/pathpolicy"
	"github.com/flopwire/flopwire/internal/transcript"
	"path/filepath"
	"testing"
)

func TestAuditSyncOnlyMetadataUnavailable(t *testing.T) {
	p := filepath.Join(t.TempDir(), "index.db")
	s, err := localindex.Open(p, localindex.Options{SyncOnly: true})
	if err != nil {
		t.Fatal(err)
	}
	err = s.SavePlacement(t.Context(), localindex.Placement{Agent: transcript.AgentClaude, SessionID: "audit-session", How: localindex.PlacedByWorktree, Placement: pathpolicy.Placement{Cwd: "/gone/app-wt", Worktree: "/gone/app-wt", Main: "/gone/app", Remote: "github.com/acme/app"}})
	if err != nil {
		t.Fatal(err)
	}
	kept, err := s.RepoDirs(t.Context())
	if err != nil {
		t.Fatal(err)
	}
	if len(kept) != 1 {
		t.Fatalf("retained metadata=%v", kept)
	}
	if err := s.Close(); err != nil {
		t.Fatal(err)
	}
	got := localRepoDirs(t.Context(), p)
	if len(got) != 0 {
		t.Fatalf("unexpected available metadata=%v", got)
	}
	t.Log("sync-only retains placement, but shared retriever discards it")
}
