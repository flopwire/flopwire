func TestAuditRawGenerationMismatch(t *testing.T) {
	b, _ := oracle(t)
	hit := grepFor(t, b, pat("expected 200"), format.Filters{}).Hits[0]
	original, err := b.RawAt(ctx, hit.Address)
	if err != nil {
		t.Fatal(err)
	}
	if err := os.Remove(hit.Provenance.Path); err != nil {
		t.Fatal(err)
	}
	remote := &fakeRemote{data: []byte(`{"unrelated":"latest generation record"}`)}
	b.Remote = remote
	got, err := b.RawAt(ctx, hit.Address)
	if err != nil {
		t.Fatal(err)
	}
	if string(got) == string(original) {
		t.Fatal("fixture failed")
	}
	t.Logf("address resolved unchanged; raw accepted unrelated latest-generation bytes; request=%s", remote.got)
}
func TestAuditDeletedRepoNeedsPlacements(t *testing.T) {
	w := newRepoWorld(t)
	dirs, err := w.b.Store.RepoDirs(ctx)
	if err != nil {
		t.Fatal(err)
	}
	path := filepath.Join(w.base, "app-old")
	full, err := ServerRepo(path, dirs, func(pathpolicy.Placement) bool { return true })
	if err != nil {
		t.Fatal(err)
	}
	minimal, err := ServerRepo(path, nil, func(pathpolicy.Placement) bool { return true })
	if err != nil {
		t.Fatal(err)
	}
	if len(full.Mains) <= len(minimal.Mains) {
		t.Fatalf("expected lost repository identity: full=%+v minimal=%+v", full, minimal)
	}
	t.Logf("full mode has %d repository main(s); missing metadata resolves %q as its own main", len(full.Mains), minimal.Mains)
}
